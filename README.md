SELECT F.REFERANS,
       F.ODMTARIH,
       COUNT(*) AS ADET
FROM otolive.t_oto_fatura f
WHERE F.REFERANS <> 0
  AND F.ODMTARIH >= TO_DATE('25.05.2025','DD.MM.YYYY')  -- son 3 ay için başlangıç
  AND F.ODMTARIH <= TO_DATE('25.08.2025','DD.MM.YYYY')  -- bugüne kadar
GROUP BY F.REFERANS, F.ODMTARIH
HAVING COUNT(*) > 1
ORDER BY F.ODMTARIH DESC, F.REFERANS;
