WITH PAID AS (
    SELECT
        P.CUSTOMER_NO                         AS CUSTOMER_NO,
        P.SUBSCRIBER_NAME                     AS NAME_SURNAME,
        I.PRODUCT_CODE                        AS PRODUCT,
        I.INSTITUTION_CODE                    AS INSTITUTION,
        S.SUBSCRIBER_NO                       AS SUBSCRIBER_NO,
        P.CURRENCY                            AS CURRENCY_CODE,
        TO_CHAR(P.BILL_DUE_DATE,'YYYYMM')     AS BILL_MONTH,          
        MAX(P.BILL_DUE_DATE)                  AS LAST_PAYMENT_DATE,   
        SUM(P.PAYMENT_AMOUNT)                 AS PAID_AMOUNT,         
        COUNT(*)                              AS PAID_COUNT,         

        MAX(P.CONTRACT_NO) KEEP (
            DENSE_RANK LAST ORDER BY P.BILL_DUE_DATE
        ) AS ACCOUNTING_CONTRACT_NO,

        MAX(P.RECEIPT_CODE) KEEP (
            DENSE_RANK LAST ORDER BY P.BILL_DUE_DATE
        ) AS TRANSACTION_CODE
    FROM BILL.PAYMENT P
    JOIN BILL.INSTITUTION I
      ON P.INSTITUTION_ID = I.ID
    JOIN BILL.SUBSCRIBER S
      ON S.INSTITUTION_ID           = I.ID
     AND S.INSTITUTION_ID           = P.INSTITUTION_ID
     AND S.INSTITUTION_DEBT_TYPE_ID = P.INSTITUTION_DEBT_TYPE_ID
     AND S.SUBSCRIBER_NO            = P.SUBSCRIBER_NO
    JOIN BILL.PAYMENT_ORDER POR
      ON POR.SUBSCRIBER_ID = S.ID
     AND POR.CUSTOMER_NO   = P.CUSTOMER_NO
    WHERE P.STATUS        = 'PAID'
      AND POR.STATUS      = 'ORDERED'
      AND P.CHANNEL_CODE  = '602'
      AND P.BILL_DUE_DATE >= ADD_MONTHS(TRUNC(SYSDATE, 'MM'), -5)
      AND POR.CUSTOMER_NO = 38121749
    GROUP BY
        P.CUSTOMER_NO,
        P.SUBSCRIBER_NAME,
        I.PRODUCT_CODE,
        I.INSTITUTION_CODE,
        S.SUBSCRIBER_NO,
        P.CURRENCY,
        TO_CHAR(P.BILL_DUE_DATE,'YYYYMM')
),
OVERDUE_AGG AS (
    SELECT
        POR.CUSTOMER_NO                        AS CUSTOMER_NO,
        I.PRODUCT_CODE                         AS PRODUCT,
        I.INSTITUTION_CODE                     AS INSTITUTION,
        S.SUBSCRIBER_NO                        AS SUBSCRIBER_NO,
        PB.CURRENCY                            AS CURRENCY_CODE,
        TO_CHAR(PB.BILL_DUE_DATE,'YYYYMM')     AS BILL_MONTH,

        SUM(PB.AMOUNT)                         AS NOT_PAID_AMOUNT,
        COUNT(*)                               AS NOT_PAID_COUNT
    FROM BILL.PAYABLE_BILL PB
    JOIN BILL.INSTITUTION I
      ON PB.INSTITUTION_ID = I.ID
    JOIN BILL.SUBSCRIBER S
      ON S.INSTITUTION_ID           = I.ID
     AND S.INSTITUTION_ID           = PB.INSTITUTION_ID
     AND S.INSTITUTION_DEBT_TYPE_ID = PB.INSTITUTION_DEBT_TYPE_ID
     AND S.SUBSCRIBER_NO            = PB.SUBSCRIBER_NO
    JOIN BILL.PAYMENT_ORDER POR
      ON POR.SUBSCRIBER_ID = S.ID
    WHERE PB.STATUS       = 'OVERDUE'
      AND POR.STATUS      = 'ORDERED'
      AND TO_CHAR(PB.BILL_DUE_DATE,'YYYYMM') >= TO_CHAR(ADD_MONTHS(SYSDATE, -5), 'YYYYMM')
      AND POR.CUSTOMER_NO = 38121749
    GROUP BY
        POR.CUSTOMER_NO,
        I.PRODUCT_CODE,
        I.INSTITUTION_CODE,
        S.SUBSCRIBER_NO,
        PB.CURRENCY,
        TO_CHAR(PB.BILL_DUE_DATE,'YYYYMM')
),
PAYABLE_AGG AS (
    SELECT
        POR.CUSTOMER_NO                        AS CUSTOMER_NO,
        I.PRODUCT_CODE                         AS PRODUCT,
        I.INSTITUTION_CODE                     AS INSTITUTION,
        S.SUBSCRIBER_NO                        AS SUBSCRIBER_NO,
        PB.CURRENCY                            AS CURRENCY_CODE,
        TO_CHAR(PB.BILL_DUE_DATE,'YYYYMM')     AS BILL_MONTH,

        SUM(PB.AMOUNT)                         AS PAYABLE_AMOUNT,
        COUNT(*)                               AS PAYABLE_COUNT
    FROM BILL.PAYABLE_BILL PB
    JOIN BILL.INSTITUTION I
      ON PB.INSTITUTION_ID = I.ID
    JOIN BILL.SUBSCRIBER S
      ON S.INSTITUTION_ID           = I.ID
     AND S.INSTITUTION_ID           = PB.INSTITUTION_ID
     AND S.INSTITUTION_DEBT_TYPE_ID = PB.INSTITUTION_DEBT_TYPE_ID
     AND S.SUBSCRIBER_NO            = PB.SUBSCRIBER_NO
    JOIN BILL.PAYMENT_ORDER POR
      ON POR.SUBSCRIBER_ID = S.ID
    WHERE PB.STATUS       = 'NOT_PAID'
      AND POR.STATUS      = 'ORDERED'
      AND TO_CHAR(PB.BILL_DUE_DATE,'YYYYMM') >= TO_CHAR(ADD_MONTHS(SYSDATE, -5), 'YYYYMM')
      AND POR.CUSTOMER_NO = 38121749
    GROUP BY
        POR.CUSTOMER_NO,
        I.PRODUCT_CODE,
        I.INSTITUTION_CODE,
        S.SUBSCRIBER_NO,
        PB.CURRENCY,
        TO_CHAR(PB.BILL_DUE_DATE,'YYYYMM')
),

--  🔑 Tüm kombinasyonları toplayan master key set
MASTER_KEYS AS (
    SELECT CUSTOMER_NO, PRODUCT, INSTITUTION, SUBSCRIBER_NO, CURRENCY_CODE, BILL_MONTH
      FROM PAID
    UNION
    SELECT CUSTOMER_NO, PRODUCT, INSTITUTION, SUBSCRIBER_NO, CURRENCY_CODE, BILL_MONTH
      FROM OVERDUE_AGG
    UNION
    SELECT CUSTOMER_NO, PRODUCT, INSTITUTION, SUBSCRIBER_NO, CURRENCY_CODE, BILL_MONTH
      FROM PAYABLE_AGG
)

SELECT
    K.CUSTOMER_NO,
    -- PAID varsa isim oradan gelir, yoksa null (istersen burada SUBSCRIBER tablosuna da join atabiliriz)
    P.NAME_SURNAME,
    K.PRODUCT,
    K.INSTITUTION,
    K.SUBSCRIBER_NO,
    K.CURRENCY_CODE,
    K.BILL_MONTH,

    P.LAST_PAYMENT_DATE,
    NVL(P.PAID_AMOUNT, 0)          AS PAID_AMOUNT,
    NVL(P.PAID_COUNT,  0)          AS PAID_COUNT,
    P.ACCOUNTING_CONTRACT_NO,
    P.TRANSACTION_CODE,

    NVL(OV.NOT_PAID_AMOUNT, 0)     AS NOT_PAID_AMOUNT,
    NVL(OV.NOT_PAID_COUNT,  0)     AS NOT_PAID_COUNT,
    NVL(PB.PAYABLE_AMOUNT,  0)     AS PAYABLE_AMOUNT,
    NVL(PB.PAYABLE_COUNT,   0)     AS PAYABLE_COUNT
FROM MASTER_KEYS K
LEFT JOIN PAID        P
  ON P.CUSTOMER_NO   = K.CUSTOMER_NO
 AND P.PRODUCT       = K.PRODUCT
 AND P.INSTITUTION   = K.INSTITUTION
 AND P.SUBSCRIBER_NO = K.SUBSCRIBER_NO
 AND P.CURRENCY_CODE = K.CURRENCY_CODE
 AND P.BILL_MONTH    = K.BILL_MONTH
LEFT JOIN OVERDUE_AGG OV
  ON OV.CUSTOMER_NO   = K.CUSTOMER_NO
 AND OV.PRODUCT       = K.PRODUCT
 AND OV.INSTITUTION   = K.INSTITUTION
 AND OV.SUBSCRIBER_NO = K.SUBSCRIBER_NO
 AND OV.CURRENCY_CODE = K.CURRENCY_CODE
 AND OV.BILL_MONTH    = K.BILL_MONTH
LEFT JOIN PAYABLE_AGG PB
  ON PB.CUSTOMER_NO   = K.CUSTOMER_NO
 AND PB.PRODUCT       = K.PRODUCT
 AND PB.INSTITUTION   = K.INSTITUTION
 AND PB.SUBSCRIBER_NO = K.SUBSCRIBER_NO
 AND PB.CURRENCY_CODE = K.CURRENCY_CODE
 AND PB.BILL_MONTH    = K.BILL_MONTH
ORDER BY
    K.SUBSCRIBER_NO,
    K.BILL_MONTH;
