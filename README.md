DECLARE
    CURSOR institution_cursor IS
        SELECT DISTINCT INSTITUTION, PRODUCT
        FROM PYM_PROCESS_CHANNELS
        WHERE PROCESS_CODE = 'PAYMENT_CANCEL'
          AND PRODUCT IN ('CEPTEL', 'ELEKTRİK', 'SU', 'DOĞALGAZ', 'TELEKOM');
          
BEGIN
    FOR record IN institution_cursor LOOP
        select SEQ_PYM_PROCESS_CHANNELS.nextval into vseqNo from dual;
        insert into PYM_PROCESS_CHANNELS (ID, LUC, PROCESS_CODE, PRODUCT, INSTITUTION, INSTITUTION_SERVICE_TYPE, DELETED, CHANNEL_CODE, PAYMENT_TYPE, STATUS, START_TIME, FINISH_TIME, LOG_STATUS, DESCRIPTION, CREATE_DATE, CREATE_TIME, CREATED_BY, UPDATE_DATE, UPDATE_TIME, UPDATED_BY)
        VALUES (vseqNo,0,'PAYMENT_CANCEL', record.PRODUCT, record.INSTITUTION,'ALL',record.DELETED ,'IVR', 'I',record.STATUS,record.START_TIME,record.FINISH_TIME,record.LOG_STATUS,record.DESCRIPTION,SYSDATE,TO_CHAR(SYSDATE, 'HH24:MI:SS'),'SYSTEM',null,null,null);
    END LOOP;
    
    COMMIT;
END;
/
