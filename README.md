SELECT p.*
FROM PYM_PROCESS_CHANNELS p
JOIN (
    SELECT INSTITUTION, PRODUCT, MIN(ID) AS MIN_ID
    FROM PYM_PROCESS_CHANNELS
    WHERE PROCESS_CODE = 'CANCEL_ORDER'
      AND PRODUCT IN ('CEPTEL', 'ELEKTRİK', 'SU', 'DOĞALGAZ', 'TELEKOM')
    GROUP BY INSTITUTION, PRODUCT
) grouped ON p.ID = grouped.MIN_ID;


SELECT *
FROM (
    SELECT p.*, 
           ROW_NUMBER() OVER (PARTITION BY INSTITUTION, PRODUCT ORDER BY ID) AS row_num
    FROM PYM_PROCESS_CHANNELS p
    WHERE PROCESS_CODE = 'CANCEL_ORDER'
      AND PRODUCT IN ('CEPTEL', 'ELEKTRİK', 'SU', 'DOĞALGAZ', 'TELEKOM')
) t
WHERE t.row_num = 1;
