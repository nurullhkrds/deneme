public interface EventData {
	
	public final static int AT_PAGE_ERROR = 0;

	/**
	 * Returns the last controller event code that has been set. If the Page Controller does not set
	 * the Controller Event, the User Event is returned.
	 * 
	 * @return <i>java.lang.String</i>, the last controller event code that has been set.
	 */
	public String getControllerEvent();

	/**
	 * Sets the controller event code that has to be returned to the Dispatcher. This method allows
	 * also to force the termination of the conversation from the conversation controller, using the
	 * constants: <br>
	 * ATConversationController.AT_FORCED_EOC_EVENT <br>
	 * ATConversationController.AT_FORCED_CANCEL_EVENT <br>
	 * that manage EOC and CANCEL events respectively. <br>
	 * Note: it is possible to generate such events only inside the initialize and onResume methods.
	 * The setController event cannot be used inside the server side method onexitErrorDialog.
	 * 
	 * @param newControllerEvent <i>java.lang.String</i>, specifies the controller event code that
	 *        has to be returned to the Dispatcher.
	 * @throws <i>ATScopeException</i>, when the method is invoked in a scope different form that
	 *         allowed.
	 */
	public void setControllerEvent(String newControllerEvent) throws FWScopeException;

	/**
	 * Returns the current ID of the widget that has generated the user event.
	 * 
	 * @return <i>java.lang.String</i>, the current ID of the widget that has generated the user
	 *         event.
	 */
	public String getUserEventWidgetID();

	/**
	 * Returns the generated user event.
	 * 
	 * @return <i>java.lang.String</i>, the generated user event.
	 */
	public String getUserEvent();

	/**
	 * Returns the number of messages generated by the application.
	 * 
	 * @return <i>int</i>, the number of messages.
	 */
	public int getMessageCount();

	/**
	 * Returns the number of messages generated by the application whose severity is equal or higher than the given severity.
	 * 
	 * @return <i>int</i>, the number of messages.
	 */
	public int getMessageCount(SEVERITY severity);

	/**
	 * Returns the number of errors generated by the application.
	 * 
	 * @return <i>int</i>, the number of errors.
	 */
	public int getErrorCount();

public interface DisplayContext {

	/**
	 * Clears the value of all the UIObjects of the specifies group. If group is equal to null, it
	 * clears the value od all the UIObjects.
	 * 
	 * @param group <i>java.lang.String</i>, the name of the group of UIObjects.cleared.
	 */
	void clear(String group);

	/**
	 * Verifies whether an object is contained in the Display Context or not
	 * 
	 * @param key <i>java.lang.String</i>, specifies the key of the object.
	 * @return <i>boolean</i>, the result of the verification.
	 */
	boolean containsKey(String key);

	/**
	 * Returns the specified UIObject.
	 * 
	 * @param name <i>java.lang.String</i>, specifies the name of the object.
	 * @return <i>com.accenture.nba.atservices.uiobjects.ATUIObject</i>, the object identified by
	 *         the specified name.
	 */
	IWidget get(String name);

	/**
	 * Verifies whether the Display Context has been re-created at the beginning of the display
	 * phase of the page controller, or it comes from a previous Display/Process phase.
	 * 
	 * @return <i>boolean</i>, value indicating whether the Display Context has been re-created at
	 *         the beginning of the display phase of the page controller, or it comes from a
	 *         previous Display/Process phase.
	 */
	boolean isDefaultDC();

	/**
	 * Enables/disables all the UIObjects of the specified group. If group is <i>null</i>, the
	 * property is applied to all the UIObjects of the Display Context.
	 * 
	 * @param newEnabled <i>boolean</i>, specifies whether the UIObjects have to be enabled (if
	 *        true) or not (if false).
	 * @param group <i>java.lang.String</i>, specifies the group of UIObjects.
	 */
	void setEnabled(boolean newEnabled, String group);

	/**
	 * Sets the valid property for all the UIObjects of the specified group. If group is
	 * <i>null</i>, the property is applied to all the UIObjects of the Display Context.
	 * 
	 * @param newValid <i>boolean</i>, specifies whether the UIObjects have to be valid (if true) or
	 *        not (if false).
	 * @param group <i>java.lang.String</i>, specifies the group of UIObjects.
	 */
	void setValid(boolean newValid, String group);

	/**
	 * Sets the visible property for all the UIObjects of the specified group. If group is
	 * <i>null</i>, the property is applied to all the UIObjects of the Display Context.
	 * 
	 * @param newVisible <i>boolean</i>, specifies whether the UIObjects have to be visible (if
	 *        true) or not (if false).
	 * @param group <i>java.lang.String</i>, specifies the group of UIObjects.
	 */
	void setVisible(boolean newVisible, String group);

	/**
	 * Sets the readOnly property for all the UIObjects of the specified group. If group is
	 * <i>null</i>, the property is applied to all the UIObjects of the Display Context.
	 * 
	 * @param newReadOnly <i>boolean</i>, specifies whether the UIObjects have to be read only (if
	 *        true) or not (if false).
	 * @param group <i>java.lang.String</i>, specifies the group of UIObjects.
	 */
	public abstract void setReadOnly(boolean newReadOnly, String group);

	/**
	 * Sets the rendered property for all the UIObjects of the specified group. If group is
	 * <i>null</i>, the property is applied to all the UIObjects of the Display Context.
	 * 
	 * @param newRendered <i>boolean</i>, specifies whether the UIObjects have to be rendered (if
	 *        true) or not (if false).
	 * @param group <i>java.lang.String</i>, specifies the group of UIObjects.
	 */
	public abstract void setRendered(boolean newRendered, String group);

	/**
	 * Sets the specified style property for an HTML element
	 * 
	 * @param ID <i>String</i> - id of specified HTML element
	 * @param styleProperty <i>String</i> - style property to set ("display" or "visibility")
	 * @param stylePropertyValue <i>String</i> - style property value
	 * 
	 * @deprecated use {@link DisplayContext#setHTMLElementVisible()} / {@link DisplayContext#setHTMLElementRendered() }instead.
	 */
	void setHTMLElementVisibility(String id, String styleProperty, String stylePropertyValue);


	/**
	 * Clears current HTML element style properties
	 */
	void clearHTMLElementVisibility();

	void setHTMLElementVisible(String id, boolean value);

	void setHTMLElementRendered(String id, boolean value);
	
	Iterator<String> getContextKeys();

    void setEnableAll(boolean enable);

    void setReadOnlyAll(boolean enable);
