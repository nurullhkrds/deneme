DECLARE
  -- Cursor tanımı
  CURSOR c_remote_service_log IS
    SELECT ID, VERSION, INSTITUTION_ID, SERVICE_TYPE, SUBSCRIBER_NO, LOG_DATE, DURATION, SEND_DATA, 
           RECEIVED_DATA, INSTITUTION_RETURN_CODE, BANK_RETURN_CODE, ADDITIONAL_INFO, 
           DATA_POWER_TRANSACTION_ID, CHANNEL_CODE, BRANCH_CODE, CHANNEL_TRANSACTION_ID, 
           CHANNEL_SESSION_ID, CREATE_DATE, CREATED_BY, UPDATE_DATE, UPDATED_BY
    FROM BILL."tmp_REMOTE_SERVICE_LOG";
  
  -- Record tipi tanımlama
  TYPE t_remote_service_log_rec IS RECORD (
    ID BILL.REMOTE_SERVICE_LOG.ID%TYPE,
    VERSION BILL.REMOTE_SERVICE_LOG.VERSION%TYPE,
    INSTITUTION_ID BILL.REMOTE_SERVICE_LOG.INSTITUTION_ID%TYPE,
    SERVICE_TYPE BILL.REMOTE_SERVICE_LOG.SERVICE_TYPE%TYPE,
    SUBSCRIBER_NO BILL.REMOTE_SERVICE_LOG.SUBSCRIBER_NO%TYPE,
    LOG_DATE BILL.REMOTE_SERVICE_LOG.LOG_DATE%TYPE,
    DURATION BILL.REMOTE_SERVICE_LOG.DURATION%TYPE,
    SEND_DATA BILL.REMOTE_SERVICE_LOG.SEND_DATA%TYPE,
    RECEIVED_DATA BILL.REMOTE_SERVICE_LOG.RECEIVED_DATA%TYPE,
    INSTITUTION_RETURN_CODE BILL.REMOTE_SERVICE_LOG.INSTITUTION_RETURN_CODE%TYPE,
    BANK_RETURN_CODE BILL.REMOTE_SERVICE_LOG.BANK_RETURN_CODE%TYPE,
    ADDITIONAL_INFO_1 BILL.REMOTE_SERVICE_LOG.ADDITIONAL_INFO_1%TYPE,
    DATA_POWER_TRANSACTION_ID BILL.REMOTE_SERVICE_LOG.DATA_POWER_TRANSACTION_ID%TYPE,
    CHANNEL_CODE BILL.REMOTE_SERVICE_LOG.CHANNEL_CODE%TYPE,
    BRANCH_CODE BILL.REMOTE_SERVICE_LOG.BRANCH_CODE%TYPE,
    CHANNEL_TRANSACTION_ID BILL.REMOTE_SERVICE_LOG.CHANNEL_TRANSACTION_ID%TYPE,
    CHANNEL_SESSION_ID BILL.REMOTE_SERVICE_LOG.CHANNEL_SESSION_ID%TYPE,
    CREATE_DATE BILL.REMOTE_SERVICE_LOG.CREATE_DATE%TYPE,
    CREATED_BY BILL.REMOTE_SERVICE_LOG.CREATED_BY%TYPE,
    UPDATE_DATE BILL.REMOTE_SERVICE_LOG.UPDATE_DATE%TYPE,
    UPDATED_BY BILL.REMOTE_SERVICE_LOG.UPDATED_BY%TYPE
  );
  
  -- Table tipi tanımlama
  TYPE t_remote_service_log_tab IS TABLE OF t_remote_service_log_rec INDEX BY PLS_INTEGER;
  v_remote_service_log_tab t_remote_service_log_tab;

BEGIN
  -- BULK COLLECT ile verileri topluca al
  OPEN c_remote_service_log;
  LOOP
    FETCH c_remote_service_log BULK COLLECT INTO v_remote_service_log_tab LIMIT 10000;
    
    EXIT WHEN v_remote_service_log_tab.COUNT = 0;
    
    -- FORALL ile hızlı insert
    FORALL i IN 1..v_remote_service_log_tab.COUNT
      INSERT INTO BILL.REMOTE_SERVICE_LOG VALUES v_remote_service_log_tab(i);
    
    -- Her toplu işlem sonrasında commit yap.
    COMMIT;
  END LOOP;
  -- Cursor'u kapat.
  CLOSE c_remote_service_log;
END;
