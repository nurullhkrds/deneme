DECLARE
    CURSOR institution_cursor IS
        SELECT DISTINCT INSTITUTION, PRODUCT
        FROM OTOLIVE.PYM_PROCESS_CHANNELS
        WHERE PROCESS_CODE = 'PAYMENT_CANCEL'
          AND PRODUCT IN ('CEPTEL', 'ELEKTRİK', 'SU', 'DOĞALGAZ', 'TELEKOM');
          
BEGIN
    FOR record IN institution_cursor LOOP
        -- Kayıt ekleme
        INSERT INTO OTOLIVE.PYM_PROCESS_CHANNELS (PROCESS_CODE, PRODUCT, INSTITUTION, CHANNEL_CODE, PAYMENT_TYPE)
        VALUES ('PAYMENT_CANCEL', record.PRODUCT, record.INSTITUTION, 'IVR', 'I');
    END LOOP;
    
    COMMIT; -- Değişiklikleri kaydet
END;
/
