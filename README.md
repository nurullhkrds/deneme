



QUERY_NAME : GET_PRODUCT_LIST_BY_CIF_NO
DATASORUCE:CBS
QUERY_DEFINITION:SELECT
    CASE
        WHEN URUN IN ('CEPTEL', 'ELEKTRİK', 'SU', 'DOĞALGAZ','TELEKOM') THEN URUN
        ELSE 'NONE'
    END AS PRODUCT,
    COUNT(*) AS ORDER_COUNT
FROM
    OTOLIVE.T_OTO_ABONE
WHERE
    MUSTERINO = :CIF_NO
    AND STATU = 'A'
GROUP BY
    CASE
        WHEN URUN IN ('CEPTEL', 'ELEKTRİK', 'SU', 'DOĞALGAZ','TELEKOM') THEN URUN
        ELSE 'NONE'
    END
COMPONENT_NAME:HMN_PYM_BillPayments
TYPE:0
